# What never travels from this machine to the server, and is therefore never compared.
#
# Read by both deploy/push.sh (as tar --exclude-from) and deploy/verify.sh, so the thing that
# is sent and the thing that is checked cannot disagree about what counts as source.
#
# Every entry here is one of three things: a secret, something the server builds for itself, or
# scratch space belonging to one machine.

# Secrets and history.
./.git
./.env

# Built on the server by deploy/release.sh. Sending ours would be sending a Windows machine's
# idea of a Linux server's dependencies.
./node_modules
./vendor
./public/build

# Generated caches. These are *supposed* to differ: the server runs composer install --no-dev,
# so its package and service discovery lists none of phpunit, pint or larastan. Shipping ours
# would put a development machine's service providers into production and rely on the release
# script to overwrite them, which is a bet with nothing to win.
./bootstrap/cache

# Points at a Vite dev server on localhost. Harmless in production only because nothing reads
# it unless public/hot exists — which is the next line.
./public/fonts-manifest.dev.json

# A stale one of these makes every asset URL point at a Vite dev server on a laptop, which a
# customer's phone can never reach.
./public/hot

# One machine's scratch space. The server builds its own in deploy/release.sh.
./public/storage
./storage/app
./storage/logs
./storage/framework/cache
./storage/framework/sessions
./storage/framework/views
./.phpunit.result.cache

# Scratch written by the test suite, including Livewire's fake uploads. It has no business on
# a server at all, and a previous deploy carried a copy there before this line existed.
./storage/framework/testing

# A local scratch file, never part of the application.
./portal_check.html
